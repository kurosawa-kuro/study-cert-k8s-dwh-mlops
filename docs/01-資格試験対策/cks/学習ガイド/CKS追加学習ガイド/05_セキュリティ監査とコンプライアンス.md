#### 1.2 カスタムルールの作成
```yaml
# custom-rules.yaml
customRules:
  - rule: Detect Privileged Container
    desc: Detect containers running with privileged mode
    condition: >
      spawned_process and
      proc.name != "pause" and
      k8s.pod.name != "" and
      k8s.pod.privileged=true
    output: >
      Privileged container detected (user=%user.name
      command=%proc.cmdline pod=%k8s.pod.name
      container=%container.name)
    priority: WARNING
    
  - rule: Detect Sensitive Mount
    desc: Detect containers mounting sensitive paths
    condition: >
      container and
      (fd.name startswith /proc or
       fd.name startswith /dev or
       fd.name startswith /sys) and
      k8s.pod.name != ""
    output: >
      Sensitive mount detected (user=%user.name
      file=%fd.name pod=%k8s.pod.name
      container=%container.name)
    priority: CRITICAL
```

### 5. コンプライアンス自動化

#### 5.1 Policy as Code実装
```yaml
apiVersion: kyverno.io/v1
kind: ClusterPolicy
metadata:
  name: compliance-policy
spec:
  validationFailureAction: enforce
  background: true
  rules:
  - name: require-pod-security-standard
    match:
      resources:
        kinds:
        - Pod
    validate:
      message: "Pod must comply with restricted security standard"
      pattern:
        spec:
          securityContext:
            runAsNonRoot: true
            runAsUser: ">0"
            fsGroup: ">0"
          containers:
          - name: "*"
            securityContext:
              allowPrivilegeEscalation: false
              runAsNonRoot: true
              capabilities:
                drop:
                - ALL
```

### 6. インシデント対応自動化

#### 6.1 自動アラートとレスポンス
```yaml
apiVersion: v1
kind: ConfigMap
metadata:
  name: incident-response
  namespace: security
data:
  response-script.sh: |
    #!/bin/bash
    
    SEVERITY=$1
    MESSAGE=$2
    POD_NAME=$3
    NAMESPACE=$4
    
    case $SEVERITY in
      "CRITICAL")
        # クリティカルインシデント時の自動対応
        kubectl delete pod $POD_NAME -n $NAMESPACE
        kubectl create event --type=Warning --reason=SecurityIncident \
                            --message="$MESSAGE" --namespace=$NAMESPACE
        ;;
      "HIGH")
        # 高レベルインシデント時のアラート
        curl -X POST -H "Content-Type: application/json" \
             -d "{\"text\":\"Security Alert: $MESSAGE\"}" \
             $SLACK_WEBHOOK_URL
        ;;
    esac
```

#### 6.2 Playbook自動実行
```yaml
apiVersion: argoproj.io/v1alpha1
kind: Workflow
metadata:
  name: incident-response-workflow
  namespace: security
spec:
  entrypoint: incident-response
  templates:
  - name: incident-response
    steps:
    - - name: assess-threat
        template: assess-threat
    - - name: isolate-pod
        template: isolate-pod
        when: "{{steps.assess-threat.outputs.result}} == 'critical'"
    - - name: notify-team
        template: notify-team
        
  - name: assess-threat
    script:
      image: python:3.9
      command: [python]
      source: |
        import json
        import sys
        
        # 脅威レベルの評価ロジック
        threat_level = "critical"
        print(threat_level)
        
  - name: isolate-pod
    resource:
      action: create
      manifest: |
        apiVersion: networking.k8s.io/v1
        kind: NetworkPolicy
        metadata:
          name: quarantine-policy
        spec:
          podSelector:
            matchLabels:
              security.incident: "true"
          policyTypes:
          - Ingress
          - Egress
          
  - name: notify-team
    container:
      image: curlimages/curl:latest
      command: [sh, -c]
      args: ["curl -X POST -d 'Security incident detected' $SLACK_WEBHOOK_URL"]
```
